<script>

const PRIX = 2;

const JOUEURS = [
    "Jerem",
    "Pascal",
    "Nico F",
    "Momo",
    "Tristan",
    "Nico",
    "Yo",
    "Did"
];

const STORAGE_KEY = "cagnotte_billard_v1";

let data = JSON.parse(
    localStorage.getItem(STORAGE_KEY)
) || {
    history: []
};

function save() {
    localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(data)
    );
}

function addDefeat(player) {

    data.history.push({
        player: player,
        amount: PRIX,
        date: new Date().toLocaleString("fr-FR")
    });

    save();

    render();

    showToast(
        player + " : +" + PRIX + " €"
    );
}

function undo() {

    if (data.history.length === 0) {
        showToast("Aucune défaite à annuler");
        return;
    }

    data.history.pop();

    save();

    render();

    showToast("Dernière défaite annulée");
}

function resetAll() {

    if (!confirm(
        "Voulez-vous vraiment remettre la cagnotte à zéro ?"
    )) {
        return;
    }

    data.history = [];

    save();

    render();

    showToast("Cagnotte remise à zéro");
}

function getCount(player) {

    return data.history.filter(
        item => item.player === player
    ).length;
}

function renderPlayers() {

    const container =
        document.getElementById("players");

    container.innerHTML = "";

    JOUEURS.forEach(player => {

        const count = getCount(player);

        const money = count * PRIX;

        const div =
            document.createElement("div");

        div.className = "player";

        div.innerHTML = `

            <div class="player-name">
                ${player}
            </div>

            <div class="player-count">
                ${count} défaite${count > 1 ? "s" : ""}
            </div>

            <div class="player-bottom">

                <div class="player-money">
                    ${money} €
                </div>

                <button
                    class="add-button"
                    onclick="addDefeat('${player}')">
                    +${PRIX} €
                </button>

            </div>

        `;

        container.appendChild(div);

    });
}

function renderHistory() {

    const container =
        document.getElementById("history");

    container.innerHTML = "";

    if (data.history.length === 0) {

        container.innerHTML = `
            <div class="empty">
                Aucune défaite enregistrée
            </div>
        `;

        return;
    }

    [...data.history]
        .reverse()
        .forEach(item => {

            const div =
                document.createElement("div");

            div.className = "history-item";

            div.innerHTML = `

                <div>
                    <div class="history-player">
                        🎱 ${item.player}
                    </div>

                    <small style="color:#777">
                        ${item.date}
                    </small>
                </div>

                <div class="history-value">
                    +${item.amount} €
                </div>

            `;

            container.appendChild(div);

        });
}

function render() {

    const total =
        data.history.length * PRIX;

    document.getElementById("total")
        .textContent = total + " €";

    document.getElementById("defaites")
        .textContent = data.history.length;

    renderPlayers();

    renderHistory();
}

function showToast(message) {

    const toast =
        document.getElementById("toast");

    toast.textContent = message;

    toast.classList.add("show");

    setTimeout(() => {

        toast.classList.remove("show");

    }, 1800);
}

render();

</script>
